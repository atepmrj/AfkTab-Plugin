# Через сколько минут без ходьбы игрок/бот считается AFK
afk-minutes: 20

# Как часто проверять таймеры (в секундах)
check-interval-seconds: 5

# Формат ника в табе у AFK игрока. {player} - ник
tab-format: "&7[AFK] {player}"

# Опускать AFK игроков в самый низ таба
# (через скорборд-команду zzz_afk; если у игрока уже есть своя команда, она не трогается)
sort-to-bottom: true

# Единственный признак активности - ходьба (перемещение).
# Игрок должен сместиться минимум на столько блоков от точки, где его засекли.
# Мелкие подёргивания и лёгкий снос водой не считаются.
# Чат, клики, поворот головы, инвентарь и т.д. НЕ считаются.
move-distance: 1.5

# Заголовок/подвал таба с реальным онлайном
# Плейсхолдеры: {online} - реальный онлайн (без AFK), {afk} - сколько AFK, {total} - всего
# Если используешь плагин TAB, поставь enabled: false
tab-header-footer:
  enabled: true
  header: []
  footer:
    - "&7Онлайн: &a{online} &8(+{afk} AFK)"

# Сообщения в чат всем игрокам при смене статуса
broadcast:
  enabled: false
  afk-on: "&7{player} теперь AFK"
  afk-off: "&7{player} вернулся"

messages:
  reloaded: "&aКонфиг AfkTab перезагружен."
  no-permission: "&cНет прав."
  only-player: "Эту команду может использовать только игрок."
